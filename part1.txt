const extractDominantColor = (imgUrl) => {
  try {
    const img = new Image(); img.crossOrigin = 'Anonymous'; img.src = imgUrl;
    img.onload = () => {
      try {
        const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d');
        canvas.width = img.width; canvas.height = img.height; ctx.drawImage(img, 0, 0);
        const data = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
        let r = 0, g = 0, b = 0, count = 0;
        for (let i = 0; i < data.length; i += 40) {
          const alpha = data[i + 3]; if (alpha < 128) continue;
          const brightness = (data[i] + data[i + 1] + data[i + 2]) / 3; if (brightness > 200) continue;
          r += data[i]; g += data[i + 1]; b += data[i + 2]; count++;
        }
        if (count > 0) {
          r = Math.floor(r / count); g = Math.floor(g / count); b = Math.floor(b / count);
          document.documentElement.style.setProperty('--primary-color', `rgb(${r}, ${g}, ${b})`);
          document.documentElement.style.setProperty('--primary-light', `rgba(${r}, ${g}, ${b}, 0.1)`);
          document.documentElement.style.setProperty('--primary-dark', `rgb(${Math.max(0, r - 30)}, ${Math.max(0, g - 30)}, ${Math.max(0, b - 30)})`);

      } catch (err) {
        document.documentElement.style.setProperty('--primary-color', '#3A2A20');
        document.documentElement.style.setProperty('--primary-light', 'rgba(58, 42, 32, 0.1)');
        document.documentElement.style.setProperty('--primary-dark', '#2A1F17');
      }
    };
    img.onerror = () => {

function crc16(str) {
  let crc = 0xFFFF;
  for (let c = 0; c < str.length; c++) {
    crc ^= str.charCodeAt(c) << 8;
    for (let i = 0; i < 8; i++) {
      if (crc & 0x8000) crc = (crc << 1) ^ 0x1021; else crc <<= 1;
    }
  }
  return (crc & 0xFFFF).toString(16).toUpperCase().padStart(4, '0');
}

function generateDynamicQRIS(staticString, amount) {
  try {
    if (!staticString) return null;
    let str = staticString.trim();
    if (str.includes('010211')) str = str.replace('010211', '010212');
    str = str.replace(/54\d{2}\d+(?=58)/, ''); str = str.replace(/6304[0-9A-Fa-f]{4}$/, '');
    const amountStr = Math.round(amount).toString(); const amountLen = amountStr.length.toString().padStart(2, '0');
    const amountTag = `54${amountLen}${amountStr}`; const insertIndex = str.indexOf('5802');
    if (insertIndex === -1) return null;
    const finalStrWithTag = str.slice(0, insertIndex) + amountTag + str.slice(insertIndex);
    const toCalculateCRC = finalStrWithTag + '6304'; const crc = crc16(toCalculateCRC);
    return toCalculateCRC + crc;
  } catch (e) { return null; }
}
